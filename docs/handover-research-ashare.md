```python
# fbp_full_grid.py 核心逻辑 (完整版在 C:\Users\j6752\src\superagent-sandbox\scripts\fbp_full_grid.py)
# 信号: 昨首板 & 回调-5~3% & 评分>=12
first_board = (chg.shift(1) >= 0.098) & (chg.shift(2) < 0.098) & (openp.shift(1) < close.shift(1))
pullback = (chg < 0.098) & (chg >= -0.05) & (chg <= 0.03)
shrink = vol < v5 * 0.8; mild = (chg >= -0.03) & (chg <= 0.03)
score = shrink*4 + mild*4 + emo + hot   # emo/hot 见 2.1 定义
sigB = first_board & pullback & (score >= 12) & cont & trading

def simulate(sg, maxd=8, tp=0.04, sl=-0.02):
    # T+1: k 从 1 起 (入场日禁卖)
    for k in range(1, maxd+1):
        j = i + k
        bars = get_min(code, str(dates[j].date()))
        if chg.iloc[j] >= 0.098: continue          # 涨停骑乘
        for p in bars:
            if p >= o_day*(1+tp): return tp        # 冲高卖
            if p <= entry*(1+sl): return sl        # 止损
        if c_day/entry - 1 <= sl: return sl        # 收盘判止损
        if k == maxd: return c_day/entry - 1       # 最长8日

def true_nav(trades, K=2):  # 真资金约束: 日级NAV+槽位+现金
    cash = 1.0; pos = {}
    for i in range(start_i, len(dates)):
        for tid in list(pos.keys()):               # 到期结算
            if i >= pos[tid][1]:
                cash += (1/K)*(1+pos[tid][2]); del pos[tid]
        while tq and tq[0][0] <= i:                # 开仓(有现金+空槽)
            if tq[0][0]==i and len(pos)<K and cash>=1/K:
                cash -= 1/K; pos[id(tq[0])] = tq.pop(0)
```
